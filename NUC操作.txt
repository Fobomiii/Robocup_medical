{

    def mono(self):
        return time.monotonic() - self.start_mono

    def emit(self, kind, **fields):
        record = {
            "wall_ns": time.time_ns(),
            "mono_s": round(self.mono(), 6),
            "kind": kind,
            "task_state": self.task_state,
        }
        record.update(fields)
        self.output.write(
            json.dumps(
                record,
                ensure_ascii=False,
                separators=(",", ":"),
            )
            + "\n"
        )

    def make_twist_callback(self, layer, topic):
        def callback(message):
            now = time.monotonic()
            previous = self.last_rx[layer]
            dt = None if previous is None else now - previous

            vx = float(message.linear.x)
            vy = float(message.linear.y)
            wz = float(message.angular.z)
            speed = math.hypot(vx, vy)

            self.last_rx[layer] = now
            self.last_speed[layer] = speed

            if self.gap_start[layer] is not None:
                duration = now - self.gap_start[layer]
                self.max_gap[layer] = max(
                    self.max_gap[layer],
                    duration,
                )
                self.emit(
                    "gap_end",
                    layer=layer,
                    topic=topic,
                    duration_s=round(duration, 6),
                    resumed_speed=round(speed, 6),
                )
                print(
                    f"\n[GAP END] {layer} "
                    f"duration={duration:.3f}s "
                    f"speed={speed:.3f}",
                    flush=True,
                )
                self.gap_start[layer] = None

            self.emit(
                "twist",
                layer=layer,
                topic=topic,
                dt_s=None if dt is None else round(dt, 6),
                vx=round(vx, 6),
                vy=round(vy, 6),
                wz=round(wz, 6),
                speed=round(speed, 6),
            )

        return callback

    def make_string_callback(self, label, topic):
        def callback(message):
            self.emit(
                "string",
                label=label,
                topic=topic,
                data=message.data,
            )

            if label == "NAV_STATUS":
                try:
                    payload = json.loads(message.data)
                    self.navigator_payload = payload
                    self.navigator_following = (
                        int(payload.get("state", -1)) == 2
                        and bool(payload.get("busy", False))
                    )
                except Exception as error:
                    self.emit(
                        "parse_error",
                        label=label,
                        error=str(error),
                    )

            elif label == "BRIDGE_CMD":
                try:
                    payload = json.loads(message.data)
                    sent = payload.get("sent_cmd") or {}
                    self.sent_speed = float(
                        sent.get("speed", 0.0)
                    )
                except Exception:
                    pass

        return callback

    def task_callback(self, message):
        current = int(message.data)
        if current != self.task_state:
            previous = self.task_state
            self.task_state = current
            self.emit(
                "task_state",
                previous=previous,
                current=current,
            )
            print(
                f"\n[TASK] {previous} -> {current}",
                flush=True,
            )
        else:
            self.task_state = current

    def odom_callback(self, label, topic, message):
        vx = float(message.twist.twist.linear.x)
        vy = float(message.twist.twist.linear.y)
        wz = float(message.twist.twist.angular.z)
        speed = math.hypot(vx, vy)

        if label == "ODOM":
            self.odom_speed = speed
        else:
            self.wheel_speed = speed

        self.emit(
            "odometry",
            label=label,
            topic=topic,
            vx=round(vx, 6),
            vy=round(vy, 6),
            wz=round(wz, 6),
            speed=round(speed, 6),
        )

    def make_cloud_callback(self, topic):
        def callback(message):
            stamp_ns = (
                int(message.header.stamp.sec) * 1_000_000_000
                + int(message.header.stamp.nanosec)
            )
            now_ns = self.get_clock().now().nanoseconds
            age_s = (now_ns - stamp_ns) / 1_000_000_000.0
            self.cloud_age[topic] = age_s

            self.emit(
                "pointcloud_header",
                topic=topic,
                frame_id=message.header.frame_id,
                stamp_ns=stamp_ns,
                age_s=round(age_s, 6),
                width=int(message.width),
                height=int(message.height),
                point_step=int(message.point_step),
            )

        return callback

    def path_callback(self, topic, message):
        length = 0.0
        for first, second in zip(
            message.poses,
            message.poses[1:],
        ):
            dx = (
                second.pose.position.x
                - first.pose.position.x
            )
            dy = (
                second.pose.position.y
                - first.pose.position.y
            )
            length += math.hypot(dx, dy)

        first_xy = None
        last_xy = None

        if message.poses:
            first = message.poses[0].pose.position
            last = message.poses[-1].pose.position
            first_xy = [
                round(first.x, 4),
                round(first.y, 4),
            ]
            last_xy = [
                round(last.x, 4),
                round(last.y, 4),
            ]

        self.emit(
            "path",
            topic=topic,
            frame_id=message.header.frame_id,
            poses=len(message.poses),
            length_m=round(length, 4),
            first_xy=first_xy,
            last_xy=last_xy,
        )

    def speed_limit_callback(self, message):
        self.emit(
            "speed_limit",
            percentage=bool(message.percentage),
            speed_limit=round(
                float(message.speed_limit),
                6,
            ),
        )

    def action_status_callback(self, message):
        statuses = []
        for item in message.status_list:
            statuses.append(
                {
                    "status": int(item.status),
                    "goal_id": bytes(
                        item.goal_info.goal_id.uuid
                    ).hex(),
                }
            )

        self.emit(
            "action_status",
            statuses=statuses,
        )

    def timer_callback(self):
        now = time.monotonic()
        active = (
            self.navigator_following
            and self.task_state in NAVIGATION_TASK_STATES
        )

        if active != self.motion_active:
            previous = self.motion_active
            self.motion_active = active

            if active:
                self.motion_start_mono = now
                self.emit(
                    "motion_start",
                    previous=previous,
                    navigator=self.navigator_payload,
                )
                print(
                    "\n========== MOTION START ==========",
                    flush=True,
                )
            else:
                self.emit(
                    "motion_end",
                    previous=previous,
                    navigator=self.navigator_payload,
                )
                print(
                    "\n========== MOTION END ==========",
                    flush=True,
                )

                for layer in TWIST_TOPICS:
                    if self.gap_start[layer] is not None:
                        duration = (
                            now - self.gap_start[layer]
                        )
                        self.max_gap[layer] = max(
                            self.max_gap[layer],
                            duration,
                        )
                        self.emit(
                            "gap_end",
                            layer=layer,
                            topic=TWIST_TOPICS[layer],
                            duration_s=round(
                                duration,
                                6,
                            ),
                            reason="motion_end",
                        )
                        self.gap_start[layer] = None

        if active and self.motion_start_mono is not None:
            for layer, topic in TWIST_TOPICS.items():
                last = self.last_rx[layer]
                reference = self.motion_start_mono

                if (
                    last is not None
                    and last >= self.motion_start_mono
                ):
                    reference = last

                age = now - reference

                if (
                    age >= GAP_THRESHOLD_S
                    and self.gap_start[layer] is None
                ):
                    self.gap_start[layer] = reference
                    self.gap_count[layer] += 1
                    self.emit(
                        "gap_start",
                        layer=layer,
                        topic=topic,
                        age_s=round(age, 6),
                    )
                    print(
                        f"\n[GAP START] "
                        f"{layer} age={age:.3f}s",
                        flush=True,
                    )

        if self.ready and now - self.last_console >= 0.5:
            self.last_console = now
            fields = []

            for layer in TWIST_TOPICS:
                last = self.last_rx[layer]
                age = (
                    None
                    if last is None
                    else now - last
                )
                speed = self.last_speed[layer]

                if age is None or speed is None:
                    fields.append(f"{layer}=--/--")
                else:
                    fields.append(
                        f"{layer}={speed:.3f}"
                        f"@{age:.2f}s"
                    )

            nav_state = self.navigator_payload.get(
                "state",
                "--",
            )
            nav_request = self.navigator_payload.get(
                "request_id",
                "--",
            )
            nav_goal = self.navigator_payload.get(
                "goal_id",
                "--",
            )
            controller = self.navigator_payload.get(
                "controller_id",
                "--",
            )

            wheel = (
                "--"
                if self.wheel_speed is None
                else f"{self.wheel_speed:.3f}"
            )
            sent = (
                "--"
                if self.sent_speed is None
                else f"{self.sent_speed:.3f}"
            )
            nav_cloud = self.cloud_age["/livox/lidar_nav"]
            safety_cloud = self.cloud_age["/livox/lidar_safety"]
            nav_cloud_text = "--" if nav_cloud is None else f"{nav_cloud:.3f}s"
            safety_cloud_text = (
                "--" if safety_cloud is None else f"{safety_cloud:.3f}s"
            )

            timestamp = datetime.now().strftime(
                "%H:%M:%S.%f"
            )[:-3]

            print(
                f"{timestamp} TASK={self.task_state} "
                + " ".join(fields)
                + f" SENT={sent}"
                + f" WHEEL={wheel}"
                + f" CLOUD_NAV={nav_cloud_text}"
                + f" CLOUD_SAFETY={safety_cloud_text}"
                + f" NAV[state={nav_state}"
                + f" req={nav_request}"
                + f" goal={nav_goal}"
                + f" controller={controller}]",
                flush=True,
            )

    def close(self):
        self.emit(
            "recorder_stop",
            gap_count=self.gap_count,
            max_gap_s={
                key: round(value, 6)
                for key, value in self.max_gap.items()
            },
        )
        self.output.close()
